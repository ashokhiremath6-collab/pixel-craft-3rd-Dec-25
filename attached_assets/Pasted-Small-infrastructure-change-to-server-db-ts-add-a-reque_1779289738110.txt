// Temporary smoke test for requestDb() and txStorage from server/db.ts.
// This file will be deleted at the end of this session.
//
// What it proves:
//   1. requestDb() returns the module-level db when called outside any
//      txStorage.run() context.
//   2. requestDb() returns the value passed into txStorage.run() when
//      called inside that async context.
//   3. The context is isolated — once txStorage.run() exits, requestDb()
//      returns to the module-level db.
//
// This does NOT touch the database. It is a pure JavaScript-runtime test
// of the AsyncLocalStorage plumbing.

import { db, txStorage, requestDb } from "../server/db";

function assert(condition: boolean, message: string): void {
  if (!condition) {
    console.error(`FAIL: ${message}`);
    process.exit(1);
  }
  console.log(`PASS: ${message}`);
}

async function main() {
  // Test 1: outside any run() context, requestDb() returns db.
  const outsideValue = requestDb();
  assert(outsideValue === db, "requestDb() returns module-level db outside txStorage.run()");

  // Test 2: inside run(), requestDb() returns the stored value.
  // We use a sentinel object instead of a real transaction client, because
  // this test should not open a real DB transaction.
  const fakeTx = { __sentinel: "fake-tx" } as any;

  await new Promise<void>((resolve) => {
    txStorage.run(fakeTx, async () => {
      const insideValue = requestDb();
      assert(insideValue === fakeTx, "requestDb() returns stored value inside txStorage.run()");
      assert(insideValue !== db, "Stored value is not the module-level db");

      // Test 3: nested awaits preserve the context.
      await new Promise((r) => setImmediate(r));
      const afterAwait = requestDb();
      assert(afterAwait === fakeTx, "Context survives setImmediate await");

      resolve();
    });
  });

  // Test 4: after run() exits, requestDb() returns to db.
  const afterRun = requestDb();
  assert(afterRun === db, "requestDb() returns to module-level db after txStorage.run() exits");

  console.log("\nAll tests passed.");
  process.exit(0);
}

main().catch((err) => {
  console.error("Test errored:", err);
  process.exit(1);
});
------------------- END scripts/test-requestdb.ts -------------------
