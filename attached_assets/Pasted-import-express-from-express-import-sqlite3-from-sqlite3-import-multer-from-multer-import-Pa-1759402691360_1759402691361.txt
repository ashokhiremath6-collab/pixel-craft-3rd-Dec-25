import express from "express";
import sqlite3 from "sqlite3";
import multer from "multer";
import Papa from "papaparse";
import fs from "fs";
import cors from "cors";

const app = express();
const port = 3000;

// Enable CORS
app.use(cors());
app.use(express.json());

// Setup SQLite DB
const db = new sqlite3.Database("./database.db");

// Create tables if not exists
db.run(`CREATE TABLE IF NOT EXISTS Projects (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT
)`);

db.run(`CREATE TABLE IF NOT EXISTS Tasks (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  project_id INTEGER,
  name TEXT,
  start_date TEXT,
  end_date TEXT,
  duration INTEGER,
  predecessor TEXT,
  responsible_person TEXT,
  approval_required TEXT,
  status TEXT,
  FOREIGN KEY(project_id) REFERENCES Projects(id)
)`);

// Multer setup for file upload
const upload = multer({ dest: "uploads/" });

// Upload tasks via CSV
app.post("/api/projects/:id/upload-tasks", upload.single("file"), (req, res) => {
  const file = fs.readFileSync(req.file.path, "utf8");

  Papa.parse(file, {
    header: true,
    complete: (results) => {
      const projectId = req.params.id;
      const stmt = db.prepare(`
        INSERT INTO Tasks 
        (project_id, name, start_date, end_date, duration, predecessor, responsible_person, approval_required, status) 
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
      `);

      results.data.forEach((row) => {
        if (row["Task Name"]) {
          stmt.run(
            projectId,
            row["Task Name"],
            row["Start Date"],
            row["End Date"],
            row["Duration"],
            row["Predecessor"],
            row["Responsible Person"],
            row["Approval Required"],
            row["Status"]
          );
        }
      });

      stmt.finalize();
      res.json({ message: "Tasks uploaded successfully" });
    },
  });
});

// Fetch tasks
app.get("/api/projects/:id/tasks", (req, res) => {
  db.all("SELECT * FROM Tasks WHERE project_id = ?", [req.params.id], (err, rows) => {
    if (err) return res.status(500).json({ error: err.message });
    res.json(rows);
  });
});

app.listen(port, () => {
  console.log(`Server running at http://localhost:${port}`);
});
